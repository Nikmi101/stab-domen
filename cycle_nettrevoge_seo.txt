{% assign series_id = "https://vnutriapps.ru/blog/cycle-snizit-trevogu/#series" %}
{% assign counter = 0 %}

{% for post in site.posts %}
  {% if post.published == false %}
    {% continue %}
  {% endif %}

  {% assign ok = false %}

  {% if post.isPartOf %}
    {% for part in post.isPartOf %}
      {% assign part_text = part | jsonify %}
      {% if part_text contains series_id %}
        {% assign ok = true %}
        {% break %}
      {% endif %}
    {% endfor %}
  {% endif %}

  {% unless ok %}
    {% continue %}
  {% endunless %}

  {% assign counter = counter | plus: 1 %}

  {% assign html = post.content %}

  {%- comment -%} =========================
      H1 (очищенный от id="...")
  ========================== {%- endcomment -%}
  {% assign h1_part = html | split: "<h1" %}
  {% assign h1_raw = h1_part[1] | split: "</h1>" | first %}

  {% assign h1_clean = h1_raw
    | remove: 'id="'
    | remove: "id="
    | split: ">"
    | last
    | strip_html
    | strip
  %}

  {% assign h1 = h1_clean %}

  {%- comment -%} =========================
      H2 LIST (очищенный)
  ========================== {%- endcomment -%}
  {% assign h2_blocks = html | split: "<h2" %}

  {% assign h2_list = "" %}

  {% for block in h2_blocks offset:1 %}
    {% assign h2_raw = block | split: "</h2>" | first %}

    {% assign h2_clean = h2_raw
      | remove: 'id="'
      | remove: "id="
      | split: ">"
      | last
      | strip_html
      | strip
    %}

    {% if h2_clean != "" %}
      {% assign h2_list = h2_list | append: " " | append: h2_clean | append: ";" %}
    {% endif %}
  {% endfor %}

{{ counter }}. Title: "{{ post.title | strip }}"
Description: "{% if post.description %}{{ post.description | strip }}{% else %}{{ post.content | strip_html | truncate: 180 }}{% endif %}"
H1: "{{ h1 }}"
H2: {{ h2_list }}
Tags: {% if post.tags %}{{ post.tags | join: ", " }}{% else %}-{% endif %}


---

{% endfor %}
